\subsection*{Sol. 1}
\begin{align*}
    \int_0^{a/2}\sqrt{2m(E_n-V_0)}dx + 
    \int_{a/2}^a\sqrt{2mE_n}dx = n\pi \hbar
\end{align*}
\begin{align*}
    \sqrt{2m(E_n-V_0)}\frac{a}{2}
    +
    \sqrt{2mE_n}\frac{a}{2}=n\pi \hbar
\end{align*}
\begin{align*}
    2m(E_n-V_0)+2mE_n
    +2\sqrt{4m^2(E_n-V_0)E_n}
    =\frac{4n^2\pi^2\hbar^2}{a^2}
\end{align*}
\begin{align*}
    2E_n-V_0
    +
    2\sqrt{(E_n-V_0)E_n}
    =4E^0_n
\end{align*}
\begin{align*}
    4(E_n-V_0)E_n = 
    4E_n^2-4E_n(4E_n^0+V_0)+(4E_n^0+V_0)^2
\end{align*}
\begin{align*}
    E_n = \frac{(4E^0_n+V_0)^2}{16 E_n^0}
\end{align*}
If $V_0$ is very small or $n$ is very large,
\begin{align*}
    \frac{(4E^0_n+V_0)^2}{16 E_n^0}\approx
    \frac{1}{16E_n^0}
    (16(E_n^0)^2+8E_n^0V_0)
    =E_n^0+\frac{V_0}{2}.
\end{align*}

\subsection*{Sol. 2}
\begin{enumerate}
    \item 
    \begin{align*}
    \frac{d^2}{dx^2}e^{if(x)/\hbar} 
    = \frac{i}{\hbar}
    f''(x)e^{if(x)/\hbar} 
    -
    \frac{1}{\hbar^2}
    (f'(x))^2e^{if(x)/\hbar} 
    = -\frac{p^2}{\hbar^2}
    e^{if(x)/\hbar} 
\end{align*}
\begin{align*}
    i\hbar f'' - (f')^2 + p^2 = 0
\end{align*}
\item 
\begin{align*}
    i\hbar(f_0''+\hbar f_1'' + \hbar^2f_2'')
    -(f_0'+\hbar f_1'+\hbar^2 f_2')^2
    +p^2 = 0
\end{align*}
\begin{align*}
    p^2 = (f_0')^2
\end{align*}
\begin{align*}
    if_0''-2f_0'f_1' = 0
\end{align*}
\begin{align*}
    if_1''-2f_0'f_2'-(f_1')^2=0
\end{align*}
\item 
\begin{align*}
    f_0 = \pm \int p
\end{align*}
\begin{align*}
    f_1' = \frac{if_0''}{2f_0'}
    \Rightarrow 
    f_1 = 
    \frac{i}{2}\log f_0'
    =\frac{i}{2}\log (\pm p)
\end{align*}
$$
f_1^+ = \frac{i}{2}\log (p)
$$
$$
f_1^- = \frac{i}{2}\log (p) - \frac{1}{2}n\pi,
$$
where $n$ is an odd number.
Thus,
\begin{align*}
    \psi(x)\approx &
    \exp\left[
    i(\pm \int p + \hbar \frac{i}{2}\log p)/\hbar
    \right]\\
    =&
    \exp\left[
    \pm \frac{i}{\hbar}\int p 
    \right]
    \exp[\frac{-1}{2}\log p]\\
    \propto&
    \frac{C}{\sqrt{|p|}}
    \exp\left[
    \pm \frac{i}{\hbar}\int p 
    \right]
\end{align*}


\end{enumerate}
