\subsection*{Sol. 1}
\begin{enumerate}
    \item 
    \begin{align*}
        \bm R + \frac{\mu}{m_1}\bm r= 
        \frac{m_1\bm r_1+m_2\bm r_2}{m_1+m_2}
        +
        \frac{m_2}{m_1+m_2}
        (\bm r_1-\bm r_2)
        =\bm r_1
    \end{align*}
    \begin{align*}
        \bm R - \frac{\mu}{m_1}\bm r= 
        \frac{m_1\bm r_1+m_2\bm r_2}{m_1+m_2}
        -
        \frac{m_2}{m_1+m_2}
        (\bm r_1-\bm r_2)
        =\bm r_2
    \end{align*}
    By the chain rule
    \begin{align*}
        \begin{pmatrix}
        \nabla_R(g\circ f) && \nabla_r(g\circ f)
        \end{pmatrix}
        =&
        \begin{pmatrix}
        \nabla_1 g && \nabla_2g
        \end{pmatrix}
        \begin{pmatrix}
            \frac{\partial f_1}{\partial R} & \frac{\partial f_1}{\partial r}\\
            \frac{\partial f_2}{\partial R} & \frac{\partial f_2}{\partial r}
        \end{pmatrix}\\
        =&
        \begin{pmatrix}
        \nabla_1 g && \nabla_2g
        \end{pmatrix}
        \begin{pmatrix}
            1 & \frac{\mu}{m_1}\\
            1 & -\frac{\mu}{m_2}
        \end{pmatrix},
    \end{align*}
    $$
    \begin{pmatrix}
            1 & \frac{\mu}{m_1}\\
            1 & -\frac{\mu}{m_2}
        \end{pmatrix}^{-1}
        =\begin{pmatrix}
            \frac{\mu}{m_2} & \frac{\mu}{m_1}\\
            1 & -1
        \end{pmatrix}
    $$
    so 
    $$
    \nabla_1 = \frac{\mu}{m_2}\nabla_R +\nabla_r
    $$
    $$
    \nabla_2 = \frac{\mu}{m_1}\nabla_R -\nabla_r.
    $$
    \item 

    \begin{align*}
        -\frac{\hbar^2}{2m_1}
        (\frac{\mu}{m_2}\nabla_R +\nabla_r)^2\psi
        -\frac{\hbar^2}{2m_2}
        (\frac{\mu}{m_1}\nabla_R -\nabla_r)^2\psi
        +V(r)\psi = E\psi
    \end{align*}
    \begin{align*}
        (-\frac{\hbar^2\mu^2}{2m_1m_2^2}
        -\frac{\hbar^2\mu^2}{2m_2m_1^2})
        \nabla^2_R
        \psi 
        +
        (-\frac{\hbar^2}{2m_1}
        -\frac{\hbar^2}{2m_2})
        \nabla_r^2 \psi 
        +V(r)\psi = E\psi.
    \end{align*}
    \begin{align*}
        -\frac{\hbar^2}{2(m_1+m_2)}
        \nabla^2_R\psi 
        -
        \frac{\hbar^2}{2\mu}
        \nabla_r^2\psi + V(r)\psi = E\psi.
    \end{align*}
    \item 
    $$
    -\frac{\hbar^2}{2(m_1+m_2)}
        \nabla^2_R\psi_R(R) = E_R\psi_R(R)
    $$
    $$
    -\frac{\hbar^2}{2\mu}
        \nabla_r^2\psi_r(r) + V(r)\psi_r(r) = E_r\psi_r(r).
    $$
\end{enumerate}
